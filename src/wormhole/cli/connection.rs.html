14pub async fn send_command(command: Command, stream: &mut Stream) -> Result<(), std::io::Error> {
15    let serialized =
16        bincode::serialize(&command).expect("Can't serialize cli command, shouldn't be possible .");
17    log::trace!("Sending cmd: {:?}, bytes {}", command, serialized.len());
18
19    stream.write_u32(serialized.len() as u32).await?;
20    stream.write_all(&serialized).await?;
21    stream.flush().await
22}

24pub async fn recieve_answer<T>(stream: &mut Stream) -> Result<T, std::io::Error>
25where
26    T: for<'a> Deserialize<'a> + std::fmt::Debug,
27{
28    let size = stream.read_u32().await?;
29    let mut recived_answer = vec![0; size as usize];
30    stream.read_exact(&mut recived_answer).await?;
31
32    bincode::deserialize::<T>(&recived_answer)
33        .map_err(|err| {
34            log::error!("Service answer deserialisation failed: {err}");
35            io::Error::new(
36                io::ErrorKind::InvalidData,
37                "Command recieved isn't recognized",
38            )
39        })
40        .inspect(|answer| log::trace!("Recieved answer: {:?}", answer))
41}
