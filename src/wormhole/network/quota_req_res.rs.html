28    fn handle_established_inbound_connection(

65    fn on_connection_handler_event(
66        &mut self,
67        peer_id: PeerId,
68        connection_id: swarm::ConnectionId,

70    ) {
71        self.request_response
72            .on_connection_handler_event(peer_id, connection_id, event);
73    }
74
75    fn poll(
76        &mut self,
77        cx: &mut std::task::Context<'_>,
78    ) -> std::task::Poll<swarm::ToSwarm<Self::ToSwarm, swarm::THandlerInEvent<Self>>> {
79        let poll = self.request_response.poll(cx);
80        if let Poll::Ready(swarm::ToSwarm::GenerateEvent(event)) = &poll {
81            match event {
82                request_response::Event::Message { message, .. } => match message {
83                    request_response::Message::Request { request_id, .. } => {
84                        self.inbound.insert(*request_id);
85                    }
86                    request_response::Message::Response { request_id, .. } => {
87                        self.permits.remove(request_id);
88                    }
89                },
90                request_response::Event::OutboundFailure { request_id, .. } => {
91                    self.permits.remove(request_id);
92                }
93                request_response::Event::InboundFailure { request_id, .. } => {
94                    self.inbound.remove(request_id);
95                }
96                request_response::Event::ResponseSent { request_id, .. } => {
97                    self.inbound.remove(request_id);
98                }
99            }
100        }
101        poll
102    }
103}

121impl<TCodec> Behaviour<TCodec>
122where
123    TCodec: Codec + Clone + Send + 'static,
124{
125    pub fn with_codec<I>(codec: TCodec, protocols: I, cfg: Config) -> Self
126    where
127        I: IntoIterator<Item = (TCodec::Protocol, ProtocolSupport)>,
128    {
129        Self {
130            request_response: request_response::Behaviour::with_codec(
131                codec,
132                protocols,
133                cfg.with_max_concurrent_streams(
134                    MAX_CONCURRENT_CONNECTIONS + MAX_CONCURRENT_STREAMS_OVERHEAD,
135                ),
136            ),
137            permits: Default::default(),
138            semaphore: Arc::new(Semaphore::new(MAX_CONCURRENT_CONNECTIONS)),
139            inbound: Default::default(),
140        }
141    }

144/// Implementations of semaphore limiting.
145impl<TCodec> Behaviour<TCodec>
146where
147    TCodec: Codec + Clone + Send + 'static,
148{
149    pub fn send_request(
150        &mut self,
151        permit: OwnedSemaphorePermit,
152        peer: &PeerId,
153        request: TCodec::Request,
154    ) -> OutboundRequestId {
155        self.send_request_with_addresses(permit, peer, request, Vec::new())
156    }
157
158    pub fn send_request_with_addresses(
159        &mut self,
160        permit: OwnedSemaphorePermit,
161        peer: &PeerId,
162        request: TCodec::Request,
163        addresses: Vec<Multiaddr>,
164    ) -> OutboundRequestId {
165        let request_id = self
166            .request_response
167            .send_request_with_addresses(peer, request, addresses);
168        self.permits.insert(request_id, permit);
169        request_id
170    }
171
172    /// cheat and generate a 0-sized permit
173    /// may only be used for special cases that can't be arbitrarily multiplied
174    pub fn nopermit(&mut self) -> OwnedSemaphorePermit {
175        Semaphore::try_acquire_many_owned(self.semaphore.clone(), 0)
176            .expect("aquiring 0 permits should never fail because the semaphore never closes")
177    }
178
179    pub fn permit(&mut self) -> Option<OwnedSemaphorePermit> {
180        Semaphore::try_acquire_owned(self.semaphore.clone()).ok()
181    }
182
183    pub fn permits(&mut self, n: u32) -> Option<Vec<OwnedSemaphorePermit>> {
184        let mut p = Semaphore::try_acquire_many_owned(self.semaphore.clone(), n).ok()?;
185        Some(
186            (0..n)
187                .map(|_| p.split(1).expect("known to be enough"))
188                .collect(),
189        )
190    }
191
192    pub fn semaphore(&self) -> Arc<Semaphore> {
193        self.semaphore.clone()
194    }
195}
