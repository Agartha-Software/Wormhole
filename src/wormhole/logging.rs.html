6pub fn custom_format(
7    fmt: &mut env_logger::fmt::Formatter,
8    record: &log::Record<'_>,
9) -> std::io::Result<()> {
10    let time = fmt.timestamp().to_string();
11    let time = time
12        .split("T")
13        .nth(1)
14        .and_then(|t| t.split("Z").next())
15        .unwrap_or(&time);
16
17    let level = record.level();
18    let level_str = level.as_str();
19    let level = match level {
20        Level::Trace => level_str.style(Style::new().cyan()),
21        Level::Debug => level_str.style(Style::new().blue()),
22        Level::Info => level_str.style(Style::new().green()),
23        Level::Warn => level_str.style(Style::new().yellow()),
24        Level::Error => level_str.style(Style::new().red().bold()),
25    };
26    let module = record
27        .module_path()
28        .and_then(|m| m.rsplit("::").next())
29        .unwrap_or("");
30
31    let module = module.bright_blue();
32    let left = "[".bright_black();
33    let right = "]".bright_black();
34
35    writeln!(
36        fmt,
37        "{left}{time} {level} {module}{right} {}",
38        record.args()
39    )
40}
